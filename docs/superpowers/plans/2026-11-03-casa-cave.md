```python
def profils_du_plat(plat):
    """Les profils d'un plat, dans l'ordre de priorité. Un dessert : « dessert » ; un accompagnement : aucun.
    Sans viande, poisson ni fromage, mais avec des légumes du rayon fruits et légumes : « legumes »."""
    if plat.get("genre") == "dessert":
        return ["dessert"]
    if plat.get("genre") == "accompagnement":
        return []
    trouves = {_PROFIL[cle_catalogue(i)] for i in plat["ingredients"] if cle_catalogue(i) in _PROFIL}
    if not trouves & _PROTEINES and any((chercher(i) or ("",))[0] == "fruits-legumes" for i in plat["ingredients"]):
        trouves.add("legumes")
    return [p for p in PROFILS if p in trouves]


def profils_du_repas(plats):
    """(profil principal, profils secondaires) d'un repas : le principal est le premier profil du premier plat de
    genre « plat » (sinon du premier plat qui en a) ; les secondaires, tous les autres profils de ses plats."""
    par_plat = [(p, profils_du_plat(p)) for p in plats]
    avec = [x for x in par_plat if x[1]]
    if not avec:
        return None, []
    principal = next((pr[0] for p, pr in avec if p.get("genre") == "plat"), avec[0][1][0])
    tous = {x for _, pr in avec for x in pr} - {principal}
    return principal, [p for p in PROFILS if p in tous]


def suggerer(cave, principal, secondaires, jour, belle=False):
    """Au plus MAX_SUGGESTIONS bouteilles pour ce repas. Écartées : vides, à garder (fenêtre pas ouverte), et les
    belles (prix_haut > SEUIL_BELLE) du dimanche au jeudi. belle=True : seulement les belles, quel que soit le jour.
    Points : +2 pour le profil principal, +1 par profil secondaire ; 0 point -> écartée.
    Tri : points, puis fin de la fenêtre la plus proche, puis le moins cher (prix inconnu en dernier), puis nom."""
    if principal is None:
        return []
    candidats = []
    for b in cave:
        est_belle = b.get("prix_haut") is not None and b["prix_haut"] > SEUIL_BELLE
        if b["quantite"] <= 0 or b["debut"] > jour.year:
            continue
        if belle and not est_belle:
            continue
        if not belle and est_belle and jour.weekday() not in _WEEK_END:
            continue
        points = 2 * (principal in b["accords"]) + sum(1 for s in secondaires if s in b["accords"])
        if points:
            candidats.append((-points, b["fin"], b.get("prix_haut") is None, b.get("prix_haut") or 0, b["nom"], b))
    candidats.sort(key=lambda c: c[:5])
    return [c[5] for c in candidats[:MAX_SUGGESTIONS]]
```

```python
MAX_TEXTE = 200


def par_defaut(couleur, millesime, annee):
    """(début, fin, accords) d'un vin ajouté sans fenêtre ni accords. Sans millésime : de cette année à +2.
    Le début n'est jamais avant cette année, la fin jamais avant le début."""
    if millesime is None:
        debut, fin = annee, annee + 2
    else:
        avant, apres = _FENETRE[couleur]
        debut, fin = max(millesime + avant, annee), millesime + apres
    return debut, max(fin, debut), list(ACCORDS_PAR_COULEUR[couleur])


def statut(b, annee):
    """« garder » (fenêtre pas encore ouverte), « vite » (elle se ferme cette année ou est passée), sinon « prete »."""
    if b["debut"] > annee:
        return "garder"
    return "vite" if b["fin"] <= annee else "prete"


def _entier(corps, cle, mini, maxi, defaut=None):
    v = corps.get(cle, defaut)
    if v is None:
        return None
    if isinstance(v, bool) or not isinstance(v, int) or not mini <= v <= maxi:
        raise ValueError(f"{cle} invalide")
    return v


def _texte(corps, cle, obligatoire=False):
    v = corps.get(cle) or ""
    if not isinstance(v, str) or len(v) > MAX_TEXTE:
        raise ValueError(f"{cle} invalide")
    v = v.strip()
    if obligatoire and not v:
        raise ValueError(f"{cle} manquant")
    return v


def valider_bouteille(corps, annee):
    """Une bouteille venue du téléphone, vérifiée et complétée (fenêtre et accords par défaut s'ils manquent)."""
    if not isinstance(corps, dict):
        raise ValueError("bouteille invalide")
    identifiant = _texte(corps, "id", obligatoire=True)
    if len(identifiant) > 64:
        raise ValueError("id invalide")
    couleur = corps.get("couleur")
    if couleur not in dict(COULEURS):
        raise ValueError("couleur invalide")
    millesime = _entier(corps, "millesime", 1900, annee)
    debut, fin, accords = par_defaut(couleur, millesime, annee)
    debut = _entier(corps, "debut", 1900, 2200, debut)
    fin = _entier(corps, "fin", 1900, 2200, fin)
    if debut > fin:
        raise ValueError("fenêtre invalide")
    accords = corps.get("accords", accords)
    if not isinstance(accords, list) or any(a not in PROFILS for a in accords):
        raise ValueError("accords invalides")
    prix_bas = _entier(corps, "prix_bas", 0, 10000)
    prix_haut = _entier(corps, "prix_haut", 0, 10000)
    if prix_bas is not None and prix_haut is not None and prix_bas > prix_haut:
        raise ValueError("prix invalides")
    return {"id": identifiant, "nom": _texte(corps, "nom", obligatoire=True), "producteur": _texte(corps, "producteur"),
            "appellation": _texte(corps, "appellation"), "region": _texte(corps, "region"), "couleur": couleur,
            "millesime": millesime, "quantite": _entier(corps, "quantite", 0, 999, 1), "debut": debut, "fin": fin,
            "prix_bas": prix_bas, "prix_haut": prix_haut, "accords": [p for p in PROFILS if p in accords],
            "note": _texte(corps, "note")}
```

Note: `accords` are returned in `PROFILS` order, so the reprise data below lists them in that order.

```python
# --- Casa Binomerie : la cave -------------------------------------------------------------------

CHAMPS_CAVE = ("nom", "producteur", "appellation", "region", "couleur", "millesime", "quantite", "debut", "fin",
               "prix_bas", "prix_haut", "accords", "note")


def lire_cave(conn):
    """Toutes les références de la cave (bues comprises), par nom."""
    with conn.cursor() as cur:
        cur.execute(f"SELECT id, {', '.join(CHAMPS_CAVE)} FROM cave ORDER BY nom, id")
        return [dict(zip(("id",) + CHAMPS_CAVE, ligne)) for ligne in cur.fetchall()]


def _ecrire_bouteille(cur, b, sur_conflit):
    valeurs = [Json(b[c]) if c == "accords" else b[c] for c in CHAMPS_CAVE]
    cur.execute(f"""INSERT INTO cave (id, {', '.join(CHAMPS_CAVE)})
                    VALUES (%s, {', '.join(['%s'] * len(CHAMPS_CAVE))}) {sur_conflit}""", [b["id"]] + valeurs)


def enregistrer_bouteille(conn, b):
    """Crée ou met à jour (même id) une référence de la cave (cave.valider_bouteille)."""
    with conn.cursor() as cur:
        _ecrire_bouteille(cur, b, "ON CONFLICT (id) DO UPDATE SET "
                          + ", ".join(f"{c} = EXCLUDED.{c}" for c in CHAMPS_CAVE) + ", maj_le = now()")
    conn.commit()


def boire(conn, identifiant):
    """Une bouteille de moins (jamais sous 0). Renvoie la nouvelle quantité, ou None si la référence n'existe pas."""
    with conn.cursor() as cur:
        cur.execute("""UPDATE cave SET quantite = GREATEST(quantite - 1, 0), maj_le = now()
                       WHERE id = %s RETURNING quantite""", (identifiant,))
        ligne = cur.fetchone()
    conn.commit()
    return ligne[0] if ligne else None


def retirer_bouteille(conn, identifiant):
    """Retire une référence de la cave. Renvoie False si elle n'existait pas."""
    with conn.cursor() as cur:
        cur.execute("DELETE FROM cave WHERE id = %s", (identifiant,))
        retiree = cur.rowcount > 0
    conn.commit()
    return retiree


def poser_cave(conn, nom_reprise, references):
    """La cave de départ (cave_depart.py), une seule fois (table reprises) : une référence déjà en base n'est pas
    touchée. Sans commit : l'appelant valide ou annule (à blanc). Renvoie le nombre de références ajoutées, ou None
    si c'est déjà fait."""
    with conn.cursor() as cur:
        cur.execute("SELECT EXISTS (SELECT 1 FROM reprises WHERE nom = %s)", (nom_reprise,))
        if cur.fetchone()[0]:
            return None
        n = 0
        for b in references:
            _ecrire_bouteille(cur, b, "ON CONFLICT (id) DO NOTHING")
            n += cur.rowcount
        cur.execute("INSERT INTO reprises (nom) VALUES (%s)", (nom_reprise,))
    return n
```
